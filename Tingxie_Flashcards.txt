export default function App() {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isFlipped, setIsFlipped] = useState(false);
  const [speechAvailable, setSpeechAvailable] = useState(true);
  const [isSpeaking, setIsSpeaking] = useState(false);
  const synthRef = useRef(window.speechSynthesis);
  const [chineseVoice, setChineseVoice] = useState(null);

  const currentWord = vocabList[currentIndex];

  // Initialize speech synthesis and find a Chinese voice
  useEffect(() => {
    if (!('speechSynthesis' in window)) {
      setSpeechAvailable(false);
      return;
    }

    const loadVoices = () => {
      const voices = synthRef.current.getVoices();
      // Try to find a Mandarin Chinese voice (zh-CN or zh-TW or just zh)
      const voice = voices.find(v => v.lang.includes('zh') || v.lang.includes('cmn')) || null;
      setChineseVoice(voice);
    };

    loadVoices();
    // Chrome sometimes loads voices asynchronously
    if (speechSynthesis.onvoiceschanged !== undefined) {
      speechSynthesis.onvoiceschanged = loadVoices;
    }
  }, []);

  const speakWord = (e) => {
    // Prevent flipping the card when clicking the speaker button
    if (e) e.stopPropagation();
    
    if (!speechAvailable || isSpeaking) return;

    // Cancel any ongoing speech
    synthRef.current.cancel();

    const utterance = new SpeechSynthesisUtterance(currentWord.word);
    if (chineseVoice) {
      utterance.voice = chineseVoice;
    }
    // Set language as fallback if specific voice isn't found
    utterance.lang = 'zh-CN'; 
    utterance.rate = 0.8; // Slightly slower for dictation

    utterance.onstart = () => setIsSpeaking(true);
    utterance.onend = () => setIsSpeaking(false);
    utterance.onerror = () => setIsSpeaking(false);

    synthRef.current.speak(utterance);
  };

  const handleNext = () => {
    setIsFlipped(false);
    synthRef.current.cancel();
    setTimeout(() => {
      setCurrentIndex((prev) => (prev + 1) % vocabList.length);
    }, 150);
  };

  const handlePrev = () => {
    setIsFlipped(false);
    synthRef.current.cancel();
    setTimeout(() => {
      setCurrentIndex((prev) => (prev - 1 + vocabList.length) % vocabList.length);
    }, 150);
  };

  const handleFlip = () => {
    setIsFlipped(!isFlipped);
  };

  return (
    <div className="min-h-screen bg-slate-900 flex flex-col items-center justify-center p-4 font-sans text-slate-100">
      
      {/* Header */}
      <div className="mb-8 text-center mt-4">
        <h1 className="text-3xl md:text-4xl font-bold text-white mb-3 tracking-wide">
          听写练习
        </h1>
        <p className="text-slate-400 font-medium text-sm md:text-base bg-slate-800/50 px-4 py-1.5 rounded-full inline-block">
          发展汉语中级 1 - บทที่ 7
        </p>
      </div>

      {/* Progress Counter */}
      <div className="bg-emerald-500/10 border border-emerald-500/20 px-5 py-2 rounded-full mb-8 shadow-sm">
        <p className="text-sm font-medium text-emerald-400 tracking-wider">
          คำที่ <span className="font-bold text-emerald-300">{currentIndex + 1}</span> / {vocabList.length}
        </p>
      </div>

      {/* Flashcard Container */}
      <div 
        className="relative w-full max-w-sm h-96 md:h-[26rem] cursor-pointer group perspective-1000 mb-8"
        onClick={handleFlip}
      >
        <div 
          className={`w-full h-full transition-transform duration-700 ease-in-out preserve-3d relative ${isFlipped ? 'rotate-y-180' : ''}`}
          style={{ transformStyle: 'preserve-3d' }}
        >
          {/* Front of Card (Tingxie Mode - Audio Only) */}
          <div 
            className="absolute w-full h-full backface-hidden bg-slate-800 rounded-3xl shadow-2xl border border-slate-700 flex flex-col items-center justify-center p-8 text-center"
            style={{ backfaceVisibility: 'hidden' }}
          >
            <span className="text-slate-500 text-sm font-medium absolute top-6 left-6">No. {currentWord.id}</span>
            
            <div className="flex-1 flex flex-col items-center justify-center w-full">
              {!speechAvailable ? (
                <div className="text-amber-400 flex flex-col items-center gap-2">
                  <VolumeX size={48} />
                  <p className="text-sm">เบราว์เซอร์ไม่รองรับเสียงอ่าน</p>
                </div>
              ) : (
                <button
                  onClick={speakWord}
                  disabled={isSpeaking}
                  className={`w-32 h-32 rounded-full flex items-center justify-center transition-all duration-300 ${
                    isSpeaking 
                      ? 'bg-emerald-500/20 text-emerald-400 scale-110 shadow-[0_0_30px_rgba(16,185,129,0.3)]' 
                      : 'bg-slate-700 text-slate-300 hover:bg-slate-600 hover:scale-105 hover:text-white shadow-xl'
                  }`}
                  aria-label="Play audio"
                >
                  <Volume2 size={64} className={isSpeaking ? 'animate-pulse' : ''} />
                </button>
              )}
              <p className="text-slate-400 mt-8 text-lg font-medium">กดเพื่อฟังเสียง</p>
            </div>

            <p className="text-xs text-slate-500 absolute bottom-6 flex items-center gap-2 bg-slate-900/50 px-4 py-2 rounded-full">
              <Eye size={14} /> แตะการ์ดเพื่อดูเฉลย
            </p>
          </div>

          {/* Back of Card (Answers) */}
          <div 
            className="absolute w-full h-full backface-hidden bg-emerald-600 rounded-3xl shadow-2xl border border-emerald-500 flex flex-col items-center justify-center p-8 text-center rotate-y-180 text-white"
            style={{ backfaceVisibility: 'hidden', transform: 'rotateY(180deg)' }}
          >
            <span className="text-emerald-200/60 text-sm font-medium absolute top-6 left-6">No. {currentWord.id}</span>
            
            <div className="bg-emerald-900/30 px-3 py-1 rounded-md text-xs font-semibold mb-4 tracking-widest border border-emerald-400/20">
              ชนิดคำ: {currentWord.type}
            </div>
            
            <div className="text-6xl md:text-7xl font-medium text-white mb-2 tracking-wider">
              {currentWord.word}
            </div>
            
            <div className="text-2xl md:text-3xl text-emerald-200 font-medium tracking-widest mb-6">
              {currentWord.pinyin}
            </div>
            
            <div className="w-full h-px bg-emerald-400/30 mb-6"></div>
            
            <div className="text-xl md:text-2xl font-bold mb-3 leading-tight text-white drop-shadow-md">
              {currentWord.thai}
            </div>
            
            <div className="text-emerald-100 text-sm md:text-base font-medium">
              {currentWord.english}
            </div>

            <p className="text-xs text-emerald-200/70 absolute bottom-6 flex items-center gap-2">
              <RotateCcw size={14} /> แตะเพื่อกลับไปฟังเสียง
            </p>
          </div>
        </div>
      </div>

      {/* Controls */}
      <div className="flex items-center gap-8">
        <button 
          onClick={handlePrev}
          className="w-14 h-14 flex items-center justify-center rounded-2xl bg-slate-800 text-slate-300 border border-slate-700 hover:bg-slate-700 hover:text-white transition-all active:scale-95 shadow-lg"
          aria-label="Previous card"
        >
          <ChevronLeft size={28} strokeWidth={2.5} />
        </button>

        <button 
          onClick={(e) => {
             // Play sound automatically on next if desired, but here we just flip it back
             handleNext();
          }}
          className="w-14 h-14 flex items-center justify-center rounded-2xl bg-slate-800 text-slate-300 border border-slate-700 hover:bg-slate-700 hover:text-white transition-all active:scale-95 shadow-lg"
          aria-label="Next card"
        >
          <ChevronRight size={28} strokeWidth={2.5} />
        </button>
      </div>

      {}
      <style dangerouslySetInnerHTML={{__html: `
        .perspective-1000 {
          perspective: 1000px;
        }
        .preserve-3d {
          transform-style: preserve-3d;
        }
        .backface-hidden {
          backface-visibility: hidden;
          -webkit-backface-visibility: hidden;
        }
        .rotate-y-180 {
          transform: rotateY(180deg);
        }
      `}} />
    </div>
  );
}
